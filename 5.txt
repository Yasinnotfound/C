

#include <stdio.h>

int num1, num2;

int main()
{
    printf("Enter your 2 numbers: ");
    scanf("%d %d", &num1, &num2);
    
    if(num1 == 0 || num2 == 0){
        printf("Zeros can't be used");
    }
   
    else if(num1 % num2 == 0){
        printf("number  %d is a multiple of number  %d", num1, num2);
    }
    
    else if(num2 % num1 == 0){
        printf("number  %d is a multiple of number  %d", num1, num2);
    }
    
    else{
        printf("Neither  of the numbers is a multiple of the other");
    }

    return 0;
}
