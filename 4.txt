

#include <stdio.h>

int entered_number, seconds, hours, minutes;

int main()
{
   printf("Enter your number:");
   scanf("%d", &entered_number);
   
   hours = entered_number / 3600; 
   minutes = (entered_number % 3600) / 60;
   seconds = entered_number % 60;
   
   printf("Hours = %d\n", hours);
   printf("Minutes = %d\n", minutes);
   printf("Seconds = %d\n", seconds);
   

    return 0;
}
