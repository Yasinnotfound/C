

#include <stdio.h>

int number;

int main()
{
    printf("Enter the number to divide by: ");
    scanf("%d", &number);
    
    if(number <= 0){
        printf("Please enter a number greater than 0");
    }
    
    for(int i = 1; i<=100;i++){
        if(i % number == 0){
            printf("%d \n", i);
        }
    }
    return 0;
}
