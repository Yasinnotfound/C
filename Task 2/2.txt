
#include <stdio.h>

int number;
unsigned long long factorial = 1;

int main() {
    
    printf("Enter a positive integer: ");
    scanf("%d", &number);
    
    if(number <= 0){
        printf("Please enter a positive integer larger than 0");
        return 1;
    }
    
    else if(number >= 20){
        printf("Invalid number, value too large. 8-byte int data type can't hold it");
        return 1;
    }
    
    for (int i = 2; i <=number; i++){
        factorial *= i;
    }
    
    printf("%d != %llu", number, factorial);

    return 0;
}
