
#include <stdio.h>

int start, end;

int main()
{
    printf("Enter your start number: ");
    scanf("%d", &start);

    printf("Enter your end number: ");
    scanf("%d", &end);

    printf("Prime numbers between %d and %d:\n", start, end);

    for(int num = start; num <=end; num++){
        if(num < 2) continue;

        int primenum = 1;
        for(int i = 2; i * i <= num; i++){
            if(num % i == 0){
                primenum = 0;
                break;
            }
        }
        if(primenum){
            printf("%d ", num);
        }
    }


    printf("\n");
    return 0;
}
