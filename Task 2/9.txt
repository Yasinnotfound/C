#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(void) {
    srand(time(NULL));
    int secret = rand() % 100 + 1;
    int guess = 0;

    while (guess != secret) {
        printf("Guess the number (1-100): ");
        scanf("%d", &guess);

        if (guess > secret)
            printf("Too high!\n");
        else if (guess < secret)
            printf("Too low!\n");
    }

    printf("You got it!\n");
    return 0;
}
